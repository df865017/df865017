```Python
class Solution:
    def lengthOfLongestSubstring(self, s: str) -> int:
        freq = {}
        right = 0
        left = 0
        best = 0
        # for i in range(len(s)):
        #     freq[s[i]] = freq.get(s[i],0) + 1
        #     # print(s[i], ":", freq[s[i]])
        for right in range(len(s)):
            ch = s[right]
            freq[ch] = freq.get(ch, 0) + 1
            while(freq[ch] > 1):
                pop_ch = s[left]
                freq[pop_ch] = freq[pop_ch] - 1
                left += 1
                # print("left:",left)
                # print("right:",right)
                # print("best:", best)
            best = max(best, right - left + 1)
        return best
      
```

##### 104、二叉树最大深度

[leetcode.cn/problems/maximum-depth-of-binary-tree](https://leetcode.cn/problems/maximum-depth-of-binary-tree/)

![1793260196416](image/1.1.2递归搜索/1793260196416.png)



```python
# Definition for a binary tree node.
# class TreeNode:
#     def __init__(self, val=0, left=None, right=None):
#         self.val = val
#         self.left = left
#         self.right = right
class Solution:
    def maxDepth(self, root: TreeNode | None) -> int:
        res1 = 1
        res2 = 1
        if (root is None): return 0
        if (root.left is not None):
            res1 = self.maxDepth(root.left) + 1
        if (root.right is not None):
            res2 = self.maxDepth(root.right) + 1
      
        return max(res1, res2)
```

```python
class Solution:

    def numIslands(self, grid: List[List[str]]) -> int:

        print("grid:", grid)
        see = [ [0] * len(row) for row in grid]
        print("see", see)


        def dfs(i, j, grid):
            if (i >=0 and i < len(grid) and 
                j >=0 and j < len(grid[0])
                ):
                if grid[i][j] == "1":
                    see[i][j] = 1
                else:
                    return
            else:
                return
            # 下
            dfs(i+1, j, grid)
            # 右
            dfs(i, j+1, grid)


        res = 0

        for i in range(len(grid)):
            for j in range(len(grid[0])):
                if (grid[i][j] == "0" and see[i][j] == 0):
                    see[i][j] = 1
                elif (grid[i][j] == "1" and see[i][j] == 0):
                    res = res + 1
                    dfs(i, j, grid)
                    print("i:j:",i,j)
                    print("see:", see)
        return res
```
